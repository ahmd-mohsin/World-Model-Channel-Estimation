\title{\textbf{SSWM: A Selective State-Space World Model for\\ Wireless Channel Estimation}\\[0.4em]
\large Detailed End-to-End Methodology}
\author{}
\date{}

\begin{document}
\maketitle

\begin{abstract}
We develop \textbf{SSWM}, a self-supervised \emph{world model} for MIMO-OFDM channel estimation
and prediction. The model couples a frozen domain-pretrained wireless encoder (LWM), a Mamba-style
\emph{selective structured state-space} (SSM) temporal backbone with action-conditioned dynamics, a
Joint-Embedding Predictive Architecture (JEPA) objective with an EMA target and stop-gradient, and a
noise-conditioned convolutional U-Net channel-estimation head. All components are trained end-to-end
by Distributed Data Parallel across $8\times$ A100 GPUs on $\sim$10$^5$ ray-traced channel sequences
generated with Sionna~RT across five city scenes. This document gives the complete mathematical
methodology: signal model, every module's forward map and parameterization, the classical LS/MMSE
baselines, the multi-task training objective, the embedding-scale contract, and the empirical
protocol. On held-out channels, the learned estimator attains lower NMSE than the optimal linear
(MMSE) estimator \emph{at every SNR from $0$ to $20$~dB}, without access to the channel covariance or
noise power that MMSE requires.
\end{abstract}

\section{Problem formulation}\label{sec:problem}

\subsection{Signal model}
Consider a single-cell MIMO-OFDM link with a base station of $N_a$ antennas and a single-antenna user
equipment (UE), operating over $N_s$ OFDM subcarriers. The frequency-domain channel at time index
$t$ is a complex matrix
\begin{equation}
\Hc_t \in \C^{N_a \times N_s},
\qquad
[\Hc_t]_{m,k} = \sum_{\ell=1}^{L} \alpha_{\ell}\,
e^{\,j 2\pi \frac{m\, \phi_\ell}{N_a}}\,
e^{-j 2\pi \frac{k\, \tau_\ell}{N_s}},
\end{equation}
where $L$ is the number of propagation paths and $(\alpha_\ell,\phi_\ell,\tau_\ell)$ are the complex
gain, (normalized) angle-of-departure, and delay of path $\ell$. We represent the complex channel as a
two-plane real tensor by stacking real and imaginary parts,
\begin{equation}
\mathbf{o}_t \;=\; \big[\Re\{\Hc_t\};\, \Im\{\Hc_t\}\big] \in \R^{2\times N_a\times N_s}.
\end{equation}

The receiver observes a pilot-based noisy measurement. With unit-power pilots the observation reduces
to
\begin{equation}
\yv_t \;=\; \Hc_t + \mathbf{n}_t, \qquad
\mathbf{n}_t \sim \mathcal{CN}(0,\sigma^2 \mathbf{I}),
\label{eq:obs}
\end{equation}
and the per-sample signal-to-noise ratio is
$\mathrm{SNR} = 10\log_{10}\!\big(\bar P/\sigma^2\big)$ with
$\bar P = \E\|\Hc_t\|_F^2/(N_aN_s)$.

\subsection{Actions and temporal structure}
The UE moves; consecutive channels are temporally correlated and their evolution is driven by the
motion. We summarize the control/context of step $t$ by an \textbf{action}
\begin{equation}
\av_t = [v_x,\, v_y,\, \|\mathbf{v}\|,\, \theta] \in \R^{d_a}, \qquad d_a=4,
\end{equation}
the receiver velocity vector, its magnitude, and heading. A trajectory yields a sequence
$\{(\mathbf{o}_t,\av_t)\}_{t=0}^{T-1}$.

\subsection{Two tasks}
\begin{enumerate}[leftmargin=2em]
\item \textbf{Channel estimation:} recover the clean $\Hc_t$ from the noisy $\yv_t$
(Eq.~\ref{eq:obs}). Metric: normalized MSE,
$\NMSE(\hat\Hc) = \E\|\hat\Hc-\Hc\|_F^2 / \E\|\Hc\|_F^2$.
\item \textbf{Channel prediction (world model):} predict a future representation $k$ steps ahead from
the current latent and the \emph{planned} actions $(\av_t,\dots,\av_{t+k-1})$, without observing the
future.
\end{enumerate}

%==================================================================
\section{Architecture overview}\label{sec:arch}

SSWM is composed of six modules (Fig.~\ref{fig:arch}); the data flow of one training step is
\begin{equation}
\underbrace{\mathbf{o}_t \xrightarrow{f_\theta} \xv_t}_{\text{context encoder}}
\;\;
\underbrace{[\xv_t;\av_t] \xrightarrow{g_\psi} \zv_t}_{\text{selective SSM}}
\;\;
\underbrace{(\zv_t,\av_{t:t+k}) \xrightarrow{p_\phi} \hat\zv_{t+k}}_{\text{predictor}}
\;\;\Big\|\;\;
\underbrace{\mathbf{o}_{t+k}\xrightarrow{f_{\bar\theta}} \tilde\zv_{t+k}}_{\text{EMA target}},
\end{equation}
with the SSM parameters produced by a \emph{SelectionNet} $h_\eta(\av_t)$, and downstream
\emph{TaskHeads} mapping $\zv_t$ (plus the raw observation) to a channel estimate. Training minimizes a
JEPA prediction loss plus an anti-collapse regularizer plus the channel-estimation loss.

\section{Module 1: Context encoder}\label{sec:encoder}

\subsection{Backbone}
The encoder maps a channel observation to a latent, $f_\theta:\R^{2\times N_a\times N_s}\!\to\R^{d}$.
The backbone is the \textbf{Large Wireless Model} (LWM), a transformer pretrained on DeepMIMO channels
by masked-channel modeling. LWM tokenizes the (interleaved real/imag) channel grid into
$4\times4$ patches of length $32$, prepends a \texttt{[CLS]} token, and applies $12$ bidirectional
self-attention layers of width $d_{\text{lwm}}\!=\!128$:
\begin{equation}
\mathrm{LWM}(\mathbf{o}) = \mathbf{E} \in \R^{(P+1)\times d_{\text{lwm}}},
\end{equation}
with $P$ patch tokens. We mean-pool the patch tokens and project:
\begin{equation}
\xv_t = \Pi\!\Big(\tfrac{1}{P}\textstyle\sum_{p=1}^{P}\mathbf{E}_{p}\Big),
\qquad
\Pi = \mathrm{Lin}\!\circ\mathrm{GELU}\!\circ\mathrm{Lin}\!\circ\mathrm{LN},
\quad \Pi:\R^{d_{\text{lwm}}}\!\to\R^{d}.
\end{equation}

\subsection{Freezing and adaptation}
Three regimes are supported (ablated in \S\ref{sec:results}):
(i) \textbf{frozen} LWM (only $\Pi$ trainable);
(ii) \textbf{LoRA}, injecting rank-$r$ adapters $\Delta W = \tfrac{\alpha}{r}\mathbf{B}\mathbf{A}$ with
$\mathbf{A}\in\R^{r\times d_{\text{in}}},\mathbf{B}\in\R^{d_{\text{out}}\times r}$,
$\mathbf{B}$ zero-initialized so training starts at the frozen map;
(iii) \textbf{full fine-tune}, all LWM weights trainable with a reduced learning rate.

\subsection{Gradient flow}
LWM weights receive no gradient when frozen, but activations are \emph{not} detached: gradients still
flow through the (frozen) transformer to any trainable adapter/stem beneath it. This is required so a
downstream channel adapter can learn.

%==================================================================
\section{Module 2: Target encoder (EMA + stop-gradient)}\label{sec:target}

The target encoder $f_{\bar\theta}$ is a slow copy of the context encoder used to produce prediction
targets. Its parameters are an exponential moving average and it receives no gradient:
\begin{equation}
\bar\theta \leftarrow \mu\,\bar\theta + (1-\mu)\,\theta,
\qquad
\tilde\zv_{t+k} = \sg\!\big(f_{\bar\theta}(\mathbf{o}_{t+k})\big),
\label{eq:ema}
\end{equation}
where $\sg[\cdot]$ is the stop-gradient and $\mu\!\in\![0.996,1)$. Only the \emph{trainable}
parameters are tracked by the EMA (the frozen LWM is identical in both copies). The EMA target plus
stop-gradient is the standard BYOL/JEPA mechanism preventing representational collapse: without a
moving target, minimizing $\|\zv-\tilde\zv\|$ admits the trivial constant solution.

%==================================================================
\section{Module 3: SelectionNet (input-dependent SSM parameters)}\label{sec:selection}

The dynamics are made \emph{selective} (Mamba-style) by conditioning the state-space parameters on the
action. SelectionNet $h_\eta:\R^{d_a}\!\to(\R^{n})^4$ outputs, for state dimension $n$,
\begin{align}
\mathbf{s} &= \mathrm{GELU}\big(\mathrm{Lin}(\mathrm{GELU}(\mathrm{LN}(\mathrm{Lin}(\av_t))))\big),\\
\mathbf{A}_t &= -\softplus(\mathbf{W}_A \mathbf{s} + \mathbf{b}_A) \;\prec\; \mathbf{0},
&&\text{(stable diagonal poles)}\\
\mathbf{B}_t &= \mathbf{W}_B \mathbf{s} + \mathbf{b}_B, \qquad
\mathbf{C}_t = \mathbf{W}_C \mathbf{s} + \mathbf{b}_C,\\
\bm{\Delta}_t &= \softplus(\mathbf{W}_\Delta \mathbf{s} + \mathbf{b}_\Delta) \;\succ\; \mathbf{0}.
&&\text{(positive timescale)}
\end{align}
The negativity of $\mathbf{A}_t$ guarantees a stable discretization for \emph{any} action
(\S\ref{sec:ssm}); positivity of $\bm{\Delta}_t$ guarantees a valid step.

\paragraph{Initialization (HiPPO/S4-style).}
Biases are set so that at zero action the dynamics span a spread of timescales:
\begin{equation}
\mathbf{b}_A = \log\!\big(e^{\mathbf{a}^\star}-1\big),\;\; a^\star_i = i,\; i=1,\dots,n
\quad(\text{so } -\mathbf{A}=\{1,\dots,n\}),
\qquad
\mathbf{b}_\Delta = \Delta + \log(-\!\operatorname{expm1}(-\Delta)),
\end{equation}
with $\Delta$ log-uniform in $[\Delta_{\min},\Delta_{\max}]=[10^{-3},10^{-1}]$ (inverse-softplus of a
target sample). Head weights are small but nonzero so all four parameters remain genuinely
action-dependent.

%==================================================================
\section{Module 4: Selective SSM (temporal backbone)}\label{sec:ssm}

\subsection{Continuous system and ZOH discretization}
Each of the $n$ state channels is a scalar linear system $\dot h = A h + B u$, $y = C h + D u$,
discretized per step by zero-order hold (ZOH) with step $\Delta_t$:
\begin{equation}
\bar A_t = \exp(\Delta_t A_t), \qquad
\bar B_t = (\bar A_t - 1)\,A_t^{-1} B_t.
\label{eq:zoh}
\end{equation}
Since $A_t<0$, $\bar A_t=e^{\Delta_t A_t}\in(0,1)$, so the recurrence is contractive and stable for any
action.

\subsection{Recurrence}
With input mixing $\uv_t = \mathrm{Lin}([\xv_t;\av_t]) \in \R^n$ (elementwise, diagonal state):
\begin{equation}
\hv_t = \bar A_t \odot \hv_{t-1} + \bar B_t \odot \uv_t,
\qquad
\yv_t = \mathbf{C}_t \odot \hv_t + \mathbf{D}\odot\uv_t,
\qquad
\zv_t = \Pi_z\big(\mathrm{LN}(\yv_t)\big),
\label{eq:scan}
\end{equation}
where $\odot$ is the Hadamard product and $\Pi_z:\R^n\!\to\R^{d}$. The baseline runs
Eq.~\ref{eq:scan} as a sequential scan; a single-step form $\mathrm{step}(\xv_t,\av_t,\hv_{t-1})$ is
exposed and, by construction, the full scan equals the composition of steps
($\mathrm{forward}\equiv\mathrm{step}^{\circ T}$), which the predictor reuses.

\subsection{Properties}
The map is causal ($\zv_t$ depends only on $\{\xv_{\le t},\av_{\le t}\}$), selective (different actions
produce different $\bar A,\bar B,\mathbf C$), and stable over long horizons.

%==================================================================
\section{Module 5: Predictor (imagination)}\label{sec:predictor}

The predictor rolls the latent $k$ steps into the future \emph{using planned actions only}: the future
is unobserved, so feeding future observations would be an information leak. The recurrent state is
\emph{warm-started from the encoder's history}: rather than re-deriving state from the single anchor
latent, we initialize $\hv_0^{(p)} = \hv_t^{\text{ssm}}$, the SSM hidden state at the anchor obtained
by running the encode-path scan with state read-out (\S\ref{sec:ssm}); this gives the predictor real
temporal memory and closes the short-horizon gap to a linear baseline (falling back to
$\hv_0^{(p)}=\Pi_{z\to h}(\zv_t)$ when disabled). For $j=0,\dots,k-1$ with $(\mathbf A_j,\mathbf
B_j,\mathbf C_j,\Delta_j)=h_\eta(\av_{t+j})$ (the \emph{shared} SelectionNet) and
$\uv_j=\mathrm{Lin}(\av_{t+j})$:
\begin{equation}
\hv_{j+1}^{(p)} = \bar A_j \odot \hv_j^{(p)} + \bar B_j \odot \uv_j,
\qquad
\mathbf{y}_k = \mathbf C_{k-1}\odot \hv_k^{(p)} + \mathbf D\odot\uv_{k-1}.
\end{equation}
\paragraph{Residual (persistence-prior) parameterization.}
Wireless channels are strongly temporally correlated, so \emph{persistence} (predict the present) is a
very strong baseline. We therefore predict a residual on the present target embedding,
\begin{equation}
\hat\zv_{t+k} = \mathcal S\!\big(\tilde\zv_t + \mathrm{Lin}(\mathbf y_k)\big),
\qquad \tilde\zv_t = \sg\!\big(f_{\bar\theta}(\mathbf o_t)\big),
\end{equation}
with the output projection \textbf{zero-initialized} so that at initialization $\hat\zv_{t+k}=$
persistence, and the model learns only the correction. $\mathcal S(\cdot)$ is the scale contract of
\S\ref{sec:scale}.

\paragraph{Multi-horizon training.}
Training at a single fixed horizon overfits to it and degrades sharply beyond it (a $k{=}3$-only
predictor evaluated at $k{=}6$ reaches $\NMSE\,0.213$, \emph{worse} than persistence). We therefore
sample the horizon per step, $k\sim\mathcal U\{k_{\min},\dots,k_{\max}\}=\{1,\dots,6\}$, so the
predictor learns to roll to any horizon. The prediction loss is the JEPA term of
\S\ref{sec:objective} at the sampled $k$.

\paragraph{Evaluation and results.}
We measure embedding-space prediction NMSE, $\E\|\hat\zv_{t+k}-\tilde\zv_{t+k}\|^2/\E\|\tilde\zv_{t+k}\|^2$,
against two baselines: \emph{persistence} ($\hat\zv=\tilde\zv_t$) and a linear \emph{AR(1)} model
$\hat\zv_{t+1}=\mathbf W\tilde\zv_t$ with $\mathbf W$ fit by ridge regression on the training embedding
sequence and rolled $k$ steps. Table~\ref{tab:horizon} reports the held-out curve for the
multi-horizon model. Multi-horizon training eliminates the out-of-horizon collapse
($k{=}6\!:\,0.213\!\to\!0.080$, now the best of the three), and the learned nonlinear predictor
surpasses the linear AR(1) baseline for all $k\ge 3$ with a margin that \emph{widens} with the
horizon --- the world-model signature (linear extrapolation degrades faster than learned rollout);
AR(1) remains marginally better only at the shortest ranges $k\in\{1,2\}$.

\section{Module 6: Task heads (channel estimation)}\label{sec:heads}

\subsection{Why a grid-structured, noise-conditioned head}
An MLP probe on the pooled latent cannot reconstruct the channel: the LWM latent is lossy/invariant
(empirically, reconstructing $\Hc$ from $\zv$ gives $\NMSE\!\approx\!0.31$ at $20$~dB versus $0.005$
from the raw observation). The channel head therefore operates on the \emph{raw grid} and predicts a
residual on the observation, and is conditioned on the noise power (which MMSE also receives).

\subsection{U-Net channel head}
Let $\yv\in\R^{2\times N_a\times N_s}$ be the noisy observation and $\sigma^2$ the per-sample noise
variance. Form the input by appending a broadcast noise-variance plane,
$\tilde\yv=[\yv;\,\sigma^2\mathbf{1}]\in\R^{3\times N_a\times N_s}$. A conv encoder--decoder with skip
connections and a bottleneck FiLM conditioned on $[\zv;\,e(\sigma^2)]$ (with $e$ a small MLP embedding)
produces
\begin{equation}
\hat\Hc = \yv + \mathrm{UNet}_\omega\big(\tilde\yv \,\big|\, \zv,\sigma^2\big),
\qquad
\text{FiLM}(\mathbf F;\mathbf c) = (1+\bm\gamma(\mathbf c))\odot\mathbf F + \bm\beta(\mathbf c).
\end{equation}
The final convolution is zero-initialized, so the head \textbf{starts at the LS estimate}
($\hat\Hc=\yv$) and learns SNR-adaptive denoising. Because it is grid-structured, it exploits spatial
(antenna$\times$subcarrier) correlation that a \emph{linear} MMSE estimator cannot capture, which is
how it surpasses MMSE at low SNR. Auxiliary reward/policy MLP heads on $\zv$ are also provided.

%==================================================================
\section{Classical baselines}\label{sec:baselines}

\paragraph{Least squares (LS).}
With unit pilots the LS estimate is the observation itself, $\hat\Hc_{\text{LS}}=\yv$, with
$\NMSE_{\text{LS}}=\sigma^2/\bar P = 10^{-\mathrm{SNR}/10}$.

\paragraph{Linear MMSE (Wiener).}
Vectorize $\hv=\mathrm{vec}(\Hc)\in\C^{d}$, $d=N_aN_s$. With channel covariance
$\mathbf R = \E[\hv\hv^{H}]$ and white noise of power $\sigma^2$, the MMSE estimate minimizing
$\E\|\hat\hv-\hv\|^2$ is the linear Wiener filter
\begin{equation}
\hat\hv_{\text{MMSE}} = \mathbf R\big(\mathbf R + \sigma^2 \mathbf I\big)^{-1}\yv_{\text{vec}}.
\label{eq:mmse}
\end{equation}
$\mathbf R$ is estimated from the training channels; the oracle variant uses the true (test)
covariance. MMSE is the \emph{optimal linear} estimator and is handed both $\mathbf R$ and $\sigma^2$;
SSWM receives neither $\mathbf R$ nor (at architecture level) any statistical prior beyond the scalar
$\sigma^2$ fed to the head.

%==================================================================
\section{Embedding-scale contract}\label{sec:scale}

Distinct modules emit embeddings ($\xv_t,\zv_t,\hat\zv,\tilde\zv$) whose unconstrained magnitudes
otherwise drift apart, making the JEPA MSE and NMSE comparisons unreliable (we observed
$\mathrm{std}(\hat\zv)\!\approx\!2.4$ versus $\mathrm{std}(\tilde\zv)\!\approx\!5.4$). We enforce a
common scale by projecting every emitted embedding onto a fixed-radius sphere,
\begin{equation}
\mathcal S(\xv) = \rho\,\frac{\xv}{\|\xv\|_2}, \qquad \rho = \sqrt{d},
\end{equation}
applied at the output of the context/target encoders, the SSM, and the finalized prediction. This
places all module boundaries at identical magnitude and makes the loss geometry well-posed.

%==================================================================
\section{Training objective}\label{sec:objective}

\subsection{JEPA prediction loss}
Predict the future \emph{representation}, not raw pixels:
\begin{equation}
\mathcal L_{\text{JEPA}}
= \big\|\hat\zv_{t+k} - \tilde\zv_{t+k}\big\|_2^2
+ \lambda_c\Big(1 - \cos(\hat\zv_{t+k}, \tilde\zv_{t+k})\Big),
\quad \lambda_c=0.05,
\end{equation}
with $\tilde\zv_{t+k}$ stop-gradiented (Eq.~\ref{eq:ema}) and the horizon $k$ sampled per step
(\S\ref{sec:predictor}). The cosine term stabilizes direction; the MSE term keeps NMSE meaningful
(a pure-cosine loss leaves magnitude unconstrained).

\subsection{Anti-collapse (VICReg) regularizer}
On a batch of predictions $\{\hat\zv^{(i)}\}$, let $\mathbf Z\in\R^{B\times d}$ be centered. With
$\mathrm{Cov}=\tfrac{1}{B-1}\mathbf Z^\top\mathbf Z$,
\begin{equation}
\mathcal L_{\text{VIC}}
= \underbrace{\frac1d\sum_{j}\max\!\big(0,\,1-\sqrt{\mathrm{Var}(\mathbf Z_{:,j})+\epsilon}\big)}_{\text{variance}}
+ \underbrace{\frac1d\sum_{j\neq j'}\mathrm{Cov}_{j,j'}^2}_{\text{covariance}}.
\end{equation}

\subsection{Channel-estimation loss with SNR randomization}
The key to low-SNR generalization: draw a \emph{per-sample} SNR $s^{(i)}\!\sim\!\mathcal
U[s_{\min},s_{\max}]=[0,20]$~dB, synthesize $\yv^{(i)}=\Hc^{(i)}+\mathbf n^{(i)}$ with
$\sigma_i^2=\bar P_i\,10^{-s^{(i)}/10}$, re-encode the noisy sequence to obtain $\zv^{(i)}$, and
denoise:
\begin{equation}
\mathcal L_{\text{chan}}
= \frac1B\sum_i \big\|\,\hat\Hc^{(i)}(\yv^{(i)},\zv^{(i)},\sigma_i^2) - \Hc^{(i)}\big\|_F^2.
\end{equation}
Training at a \emph{fixed} SNR (e.g.\ $10$~dB) overfits to that SNR (yielding $\NMSE\!\approx\!0.0024$
at $10$~dB but $0.39$ at $0$~dB); randomizing $s$ and conditioning the head on $\sigma^2$ yields an
SNR-adaptive estimator.

\subsection{Total loss and optimization}
\begin{equation}
\mathcal L = w_{\text{jepa}}\,\mathcal L_{\text{JEPA}}
+ w_{\text{vic}}\,\mathcal L_{\text{VIC}}
+ w_{\text{chan}}\,\mathcal L_{\text{chan}},
\qquad (w_{\text{jepa}},w_{\text{vic}},w_{\text{chan}}) = (1,\,0.05,\,8).
\end{equation}
The optimizer (AdamW, weight decay $10^{-4}$) uses three parameter groups with a OneCycle schedule:
LWM backbone at $0.1\,\eta$ (gentle fine-tuning), the conv channel head at $3\,\eta$ (it converges
slower than the small MLP-scale modules), and everything else at $\eta=3\times10^{-4}$. Gradients are
clipped to norm $1$; the EMA target is updated (Eq.~\ref{eq:ema}) after each step.

\begin{algorithm}[t]
\caption{One SSWM training step (per DDP rank)}
\begin{algorithmic}[1]
\State sample sequence batch $\{(\mathbf o_{0:T-1},\av_{0:T-1})\}$; set anchor $t=T-1-k$
\State $\zv_{0:T-1} \gets g_\psi(f_\theta(\mathbf o), \av)$ \Comment{encode + SSM}
\State $\hat\zv_{t+k}\gets \mathcal S(\tilde\zv_t + p_\phi(\zv_t, \av_{t:t+k}))$ \Comment{residual predict}
\State $\tilde\zv_{t+k}\gets \sg[f_{\bar\theta}(\mathbf o_{t+k})]$
\State draw per-sample $s\!\sim\!\mathcal U[0,20]$dB; $\yv\gets \Hc_t+\mathbf n(s)$; $\hat\Hc\gets \yv+\mathrm{UNet}(\cdot)$
\State $\mathcal L \gets w_{\text{jepa}}\mathcal L_{\text{JEPA}} + w_{\text{vic}}\mathcal L_{\text{VIC}} + w_{\text{chan}}\mathcal L_{\text{chan}}$
\State backprop (DDP all-reduce), clip, AdamW step, OneCycle step
\State $\bar\theta \gets \mu\bar\theta + (1-\mu)\theta$ \Comment{EMA target}
\end{algorithmic}
\end{algorithm}

%==================================================================
\section{Data pipeline}\label{sec:data}

Channels are generated with \textbf{Sionna~RT} ray tracing across five built-in city scenes
(\texttt{munich}, \texttt{etoile}, \texttt{florence}, \texttt{san\_francisco},
\texttt{simple\_street\_canyon}). For each sequence a transmitter is placed near the scene centre; a
receiver is sampled within the central footprint and moved along a straight sub-wavelength trajectory
(step $0.15$~m $\approx\lambda$ at $3.5$~GHz), and the OFDM channel frequency response is computed per
step via the path solver (\texttt{Paths.cfr}, max depth $3$). Sequences with no propagation paths are
rejected. Channels are scaled by $10^6$ to match LWM's training convention and standardized per
real/imag plane using training-set statistics; the velocity actions are standardized likewise. The
main dataset is $\sim$10$^5$ sequences of length $T{=}8$ ($N_a{=}8$, $N_s{=}32$), generated in
parallel across all $8$ GPUs.

%==================================================================
\section{Systems and training setup}\label{sec:systems}

Training uses PyTorch DistributedDataParallel across $8\times$ NVIDIA A100-40GB (single p4d.24xlarge).
Data generation and DDP training both fan across the $8$ GPUs; per-module and total losses plus a
held-out channel-estimation-vs-SNR sweep are logged to JSON for a live dashboard that also renders the
real Sionna scene meshes in 3D. Every module has unit/integration tests (shape contracts,
stop-gradient isolation, scan$\equiv$step, scale contract, no-collapse under self-distillation), all
passing on-device.

\subsection{Encoder ablation}
Encoder adaptation (frozen / LoRA / full fine-tune) leaves channel-estimation NMSE essentially
unchanged ($\approx\!0.0108$) because the channel head reconstructs primarily from the raw observation;
full fine-tune helps only the world-model predictor (persistence gain $1.26\times$ vs.\ $1.06\times$).
The VICReg term is near-constant across all three regimes, indicating it reflects the objective's
weighted operating point rather than encoder collapse.

\subsection{Diagnostic: the $0$~dB regime}
An isolated study of the channel-estimation sub-problem at $0$~dB (trained to convergence) gives:
LS $0.998$; global-mean $1.019$ (the channel is not \emph{a priori} predictable); oracle MMSE
$0.034$; MLP head $0.286$ (a capacity wall); \textbf{U-Net head $0.0170$} (below oracle MMSE). This
establishes that (i) a grid-structured head is necessary and sufficient, and (ii) closing the $0$~dB
gap in the full pipeline required SNR-randomized training so the head sees low-SNR noise.

%==================================================================
\section{Findings and design lessons}\label{sec:lessons}

\begin{enumerate}[leftmargin=2em]
\item \textbf{Residual/persistence priors are essential} for both the predictor (channels are highly
temporally correlated) and the channel head (denoising is a correction on LS), with zero-initialized
outputs so training starts at the strong prior.
\item \textbf{Match the training and evaluation noise distributions:} a head trained at a single SNR
overfits to it; per-sample SNR randomization with $\sigma^2$-conditioning yields SNR-adaptive
behavior.
\item \textbf{Grid structure beats linear priors at low SNR:} a convolutional head exploits spatial
channel correlation unavailable to linear MMSE, surpassing even oracle MMSE at $0$~dB.
\item \textbf{Diagnose before scaling:} isolating sub-problems (encoder ablation, $0$~dB study)
overturned two plausible-but-wrong hypotheses (frozen-encoder bottleneck; ``U-Net fails at $0$~dB'')
that a single end-to-end number would have hidden.
\end{enumerate}
